extends SceneTree
const SCENE: PackedScene = preload("res://aircraft/d4y2/d4y2.tscn")
const G: Script = preload("res://aircraft/d4y2/geometry.gd")

func _initialize() -> void: call_deferred("_run")

func _run() -> void:
	var model: Node3D = SCENE.instantiate() as Node3D; root.add_child(model)
	var mesh: MeshInstance3D = model.get_node("Center/ContinuousSkin") as MeshInstance3D
	var faces: PackedVector3Array = mesh.mesh.get_faces()
	var direction: Vector3 = model.get_parked_transform().basis.inverse() * Vector3(7.0, 2.0, -9.0).normalized()
	for z: float in [1.3, 1.4, 1.5, 1.6, 1.7, 1.8, 1.9]:
		var y: float = 0.02; var s: Array = G.sample(model.BODY, z)
		var x: float = s[2] * sqrt(1.0 - pow(y / float(s[3]), 2))
		var target: Vector3 = Vector3(x, y, z)
		var origin: Vector3 = target + direction * 20.0
		var hits: Array = []
		for i: int in range(0, faces.size(), 3):
			var hit: Variant = Geometry3D.ray_intersects_triangle(origin, -direction, faces[i], faces[i + 1], faces[i + 2])
			if hit != null: hits.append({"at": [hit.x, hit.y, hit.z], "distance": origin.distance_to(hit), "triangle": i / 3})
		hits.sort_custom(func(a: Dictionary, b: Dictionary) -> bool: return a.distance < b.distance)
		print("PAINT_RAY ", z, " ", JSON.stringify(hits.slice(0, 4)))
	model.free(); quit()
